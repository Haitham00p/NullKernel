#ifndef NULLOS_DRIVERS_H
#define NULLOS_DRIVERS_H

#include <stdint.h>
#include <stddef.h>

/* Lightweight driver / component initialization registry.
 *
 * Adding a driver to NullOS no longer requires editing the Makefile
 * C_SOURCES list AND wiring a call into kernel_main: drop a source file
 * that registers itself with DRIVER_INIT(...) and the kernel runs it
 * automatically during boot (in link order). This is what makes the OS
 * ready to absorb new drivers, services and hosting components without
 * touching the core startup code.
 *
 * Example (in any driver .c file):
 *     static void mydrv_init(void) { ... }
 *     DRIVER_INIT(mydrv_init)
 */
typedef void (*DriverInitFn)(void);

typedef struct
{
    const char   *name;
    DriverInitFn  fn;
} DriverEntry;

/* Place a function pointer into the .drivers.init linker section. */
#define DRIVER_INIT(fn)                                                       \
    static const DriverEntry _drv_##fn                                       \
        __attribute__((used, section(".drivers.init"))) = { #fn, fn }

/* Provided by the linker script (arch/x86_64/linker.ld). */
extern DriverEntry __init_start[];
extern DriverEntry __init_end[];

/* Run every registered initializer exactly once, in link order. */
static inline void DriversInitAll(void)
{
    for (DriverEntry *e = __init_start; e < __init_end; e++)
    {
        if (e->fn != NULL)
        {
            e->fn();
        }
    }
}

#endif
