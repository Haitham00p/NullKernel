#ifndef N64A_PORT_H
#define N64A_PORT_H

/* Portability layer for n64a.
 *
 * The assembler core must compile in three environments without changes:
 *   1. Host build   (build/host/n64a CLI, tests)  -> N64A_HOST_BUILD
 *   2. Kernel build (linked into NullOS, used by self-hosting) -> default
 *   3. NCC build    (the kernel itself may later be rebuilt with n64a parts
 *      compiled through NCC; only freestanding C constructs are used)
 */

#ifdef N64A_HOST_BUILD
#include <stdint.h>
#include <stdbool.h>
#include <stdlib.h>
#include <string.h>

#define N64A_MALLOC(SZ)  malloc((SZ) ? (SZ) : 1U)
#define N64A_FREE(P)     free((P))

static inline void *N64aMemSet(void *d, int c, unsigned long long n)
{
    return memset(d, c, (size_t)n);
}
static inline void *N64aMemCpy(void *d, const void *s, unsigned long long n)
{
    return memcpy(d, s, (size_t)n);
}
static inline unsigned long long N64aStrLen(const char *s)
{
    return (unsigned long long)strlen(s);
}
static inline int N64aStrCmp(const char *a, const char *b)
{
    return strcmp(a, b);
}
static inline void N64aStrCpy(char *d, const char *s)
{
    size_t i;
    for (i = 0; s[i] != '\0'; i++) d[i] = s[i];
    d[i] = '\0';
}
static inline void N64aStrNCpy(char *d, const char *s, unsigned long long n)
{
    unsigned long long i = 0;
    while (i + 1U < n && s[i] != '\0') { d[i] = s[i]; i++; }
    d[i] = '\0';
}

#else /* kernel / freestanding build */

#include <stdint.h>
#include <stddef.h>
#include <stdbool.h>
#include "lib/string/string.h"
#include "mm/heap/heap.h"

#define N64A_MALLOC(SZ)  kmalloc((uint64_t)(SZ))
#define N64A_FREE(P)     kfree((P))

#define N64aMemSet    memset
#define N64aMemCpy    memcpy
#define N64aStrLen    strlen
#define N64aStrCmp    strcmp

static inline void N64aStrCpy(char *d, const char *s)
{
    strcpy(d, s);
}
static inline void N64aStrNCpy(char *d, const char *s, unsigned long long n)
{
    strncpy(d, s, n);
}

#endif

#endif /* N64A_PORT_H */
