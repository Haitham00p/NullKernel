#ifndef MOUSE_H
#define MOUSE_H

#include <stdint.h>
#include <stdbool.h>

typedef struct {
    int32_t  X;      /* current pixel position */
    int32_t  Y;
    int32_t  dX;     /* last relative movement */
    int32_t  dY;
    bool     Left;
    bool     Right;
    bool     Middle;
    bool     PacketReady;
} MOUSE_STATE;

void        MouseInitialize(void);
void        MouseInterrupt(void);          /* called from IRQ12 */
bool        MouseAvailable(void);
MOUSE_STATE MouseGetState(void);

/* Draws/moves the on-screen cursor; call once per timer tick. */
void        MouseCursorUpdate(void);

#endif /* MOUSE_H */
