#ifndef NCC_H
#define NCC_H

#include <stdint.h>
#include <stdbool.h>

/* Compiles Source into a final ELF .EXE image (legacy single-file mode). */
bool NccCompile(const char *SourcePath, const char *Source,
                uint32_t SourceSize, uint8_t **OutImage, uint32_t *OutSize);

/* Compiles Source into a relocatable N64O object for mixed C/assembly
 * linking through nld.  Symbolic references become R_ABS64 relocations
 * so the linker can bind them to assembly objects or kernel symbols. */
bool NccCompileObject(const char *SourcePath, const char *Source,
                      uint32_t SourceSize, uint8_t **OutObject,
                      uint32_t *OutObjectSize);

#endif /* NCC_H */
